CREATE DATABASE empleados;
USE empleados;

CREATE TABLE emp (
	id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100),
    edad INT,
    pais VARCHAR(100),
	cargo VARCHAR(100),
    sueldo DOUBLE
)

SELECT id, nombre, edad, pais, cargo, sueldo, (sueldo * 0.18) AS igv FROM emp;

-- Procedimiento para buscar el empleado:
DELIMITER //
CREATE PROCEDURE sp_buscarempleado(in idempleado INT)
BEGIN
	SELECT nombre, edad, pais, cargo, sueldo, (sueldo * 0.18) AS igv FROM emp
    WHERE id = idempleado;
END //
DELIMITER ; 
CALL sp_buscarempleado("1");
